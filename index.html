// Чистый список приветствий в стиле OpenAI / Google Gemini const dynamicGreetingsData = { ru: [ "Что сегодня в повестке дня?", "О чем поговорим сегодня?", "Чем я могу помочь вам сегодня?", "Вам слово!", "Какую задачу решим прямо сейчас?", "Как настроение?", "Чем займемся?" ], en: [ "What's on the agenda today?", "What shall we talk about today?", "How can I help you today?", "The floor is yours!", "What are we working on today?" ], fr: [ "Qu'y a-t-il à l'ordre du jour aujourd'hui ?", "De quoi allons-nous parler aujourd'hui ?", "Comment puis-je vous aider aujourd'hui ?", "C'est à vous !" ] }; function getGreeting() { const lang = localStorage.getItem('pyorma_lang') || 'ru'; const name = localStorage.getItem('pyorma_username') || ''; const list = dynamicGreetingsData[lang] || dynamicGreetingsData['ru']; // Выбираем случайную фразу let text = list[Math.floor(Math.random() * list.length)]; // Если имя заполнено, красивой вставкой добавляем его в начало (как в Gemini) if (name) { return `${name}, ${text.toLowerCase()}`; } return text; }
